    <script>
        function toggleAuthorityFields() {
            const isDispatch = document.getElementById('auth-yes').checked;
            const dispatchFields = document.getElementById('fields-dispatch');
            const leaseFields = document.getElementById('fields-lease');

            if (isDispatch) {
                dispatchFields.style.display = 'grid';
                leaseFields.style.display = 'none';
            } else {
                dispatchFields.style.display = 'none';
                leaseFields.style.display = 'grid';
            }
        }

        function updateCalculator() {
            const miles = parseFloat(document.getElementById('miles-range').value);
            document.getElementById('miles-val').innerText = miles.toLocaleString();

            const equipType = document.getElementById('equip-type');
            const customRpmInput = document.getElementById('custom-rpm');
            
            if (document.activeElement === equipType) {
                customRpmInput.value = equipType.value;
            }

            const rpm = parseFloat(customRpmInput.value) || 0;

            const gross = miles * rpm;
            const fee = gross * 0.12;
            const fuel = (miles / 6.5) * 4.15;
            const insurance = 250;
            const net = gross - fee - fuel - insurance;

            document.getElementById('gross-revenue').innerText = '$' + gross.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
            document.getElementById('fee-deduct').innerText = '-$' + fee.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
            document.getElementById('fuel-deduct').innerText = '-$' + fuel.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
            document.getElementById('net-pay').innerText = '$' + Math.max(0, net).toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
        }

        window.onload = updateCalculator;
    </script>
